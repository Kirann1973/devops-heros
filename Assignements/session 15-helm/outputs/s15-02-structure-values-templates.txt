########## Chart structure, Chart.yaml, values.yaml and templates ##########
$ (03 simple chart: the minimum a chart needs)
$ find 03-chart-structure/simple-chart -type f | sort
03-chart-structure/simple-chart/Chart.yaml
03-chart-structure/simple-chart/templates/deployment.yaml
03-chart-structure/simple-chart/templates/service.yaml
03-chart-structure/simple-chart/values.yaml

$ helm template demo 03-chart-structure/simple-chart
---
# Source: simple-chart/templates/service.yaml
apiVersion: v1
kind: Service
metadata:
  name: demo-svc
spec:
  selector:
    app: demo
  ports:
    - port: 80
      targetPort: 80
---
# Source: simple-chart/templates/deployment.yaml
apiVersion: apps/v1
kind: Deployment
metadata:
  name: demo-app
spec:
  replicas: 1
  selector:
    matchLabels:
      app: demo
  template:
    metadata:
      labels:
        app: demo
    spec:
      containers:
        - name: app
          image: "nginx:latest"
          ports:
            - containerPort: 80

$ (05 values: defaults, then -f file, then --set (the last one wins))
$ helm template demo 05-values-yaml/my-app | grep -E 'replicas:|image:'
  replicas: 1
          image: "nginx:1.16.0"
      image: busybox

$ helm template demo 05-values-yaml/my-app -f 05-values-yaml/values-prod.yaml | grep -E 'replicas:|image:'
  replicas: 5
          image: "nginx:latest"
      image: busybox

$ helm template demo 05-values-yaml/my-app -f 05-values-yaml/values-prod.yaml --set replicaCount=2 --set image.tag=1.27 | grep -E 'replicas:|image:'
  replicas: 2
          image: "nginx:1.27"
      image: busybox
