########## Task 2: Secret - create, inject into a Pod, verify, and why it must not go to Git ##########
$ (encode values with echo -n (plain echo adds a newline))
$ echo -n "yatri_admin" | base64
eWF0cmlfYWRtaW4=

$ echo -n "secretpassword" | base64
c2VjcmV0cGFzc3dvcmQ=

$ kubectl apply -f 02-secret/db-secret.yaml
secret/yatri-db-secret created

$ kubectl get secret yatri-db-secret
NAME              TYPE     DATA   AGE
yatri-db-secret   Opaque   3      0s

$ kubectl describe secret yatri-db-secret | grep -A5 "^Data"
Data
====
POSTGRES_DB:        19 bytes
POSTGRES_PASSWORD:  14 bytes
POSTGRES_USER:      11 bytes

$ (base64 is NOT encryption: anyone who can read the YAML (or the Git repo) gets the password)
$ kubectl get secret yatri-db-secret -o jsonpath="{.data.POSTGRES_PASSWORD}"; echo
c2VjcmV0cGFzc3dvcmQ=

$ kubectl get secret yatri-db-secret -o jsonpath="{.data.POSTGRES_PASSWORD}" | base64 -d; echo
secretpassword

$ kubectl apply -f 02-secret/secret-pod.yaml
pod/secret-demo created

$ kubectl wait --for=condition=Ready pod/secret-demo --timeout=180s
pod/secret-demo condition met

$ kubectl logs secret-demo
secret-demo started by Kiran N (10446): DB user is yatri_admin

$ (inside the container the values are already decoded)
$ kubectl exec secret-demo -- env | grep DB_ | sort
DB_PASSWORD=secretpassword
DB_USER=yatri_admin

$ kubectl exec secret-demo -- ls -l /etc/db-secret/
total 0
lrwxrwxrwx    1 root     root            18 Oct 17 20:35 POSTGRES_DB -> ..data/POSTGRES_DB
lrwxrwxrwx    1 root     root            24 Oct 17 20:35 POSTGRES_PASSWORD -> ..data/POSTGRES_PASSWORD
lrwxrwxrwx    1 root     root            20 Oct 17 20:35 POSTGRES_USER -> ..data/POSTGRES_USER

$ kubectl exec secret-demo -- cat /etc/db-secret/POSTGRES_DB; echo
yatri_production_db

$ (safer way: create the Secret from the command line, never write the value in a file in Git)
$ kubectl create secret generic demo-secret --from-literal=API_KEY=kiran-10446-demo --dry-run=client -o yaml
apiVersion: v1
data:
  API_KEY: a2lyYW4tMTA0NDYtZGVtbw==
kind: Secret
metadata:
  name: demo-secret

$ kubectl delete pod secret-demo --grace-period=1
pod "secret-demo" deleted from default namespace

$ kubectl delete -f 02-secret/db-secret.yaml
secret "yatri-db-secret" deleted from default namespace
