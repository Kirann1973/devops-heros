########## TASK 1: git commit -m vs git commit -a -m ##########
$ echo "line 1" > notes.txt

$ git add notes.txt

$ git commit -m "Add notes.txt"
[main (root-commit) 3c4439e] Add notes.txt
 1 file changed, 1 insertion(+)
 create mode 100644 notes.txt

$ echo "line 2" >> notes.txt

$ echo "new file" > extra.txt

$ git status --short
 M notes.txt
?? extra.txt

########## 1a. git commit -m WITHOUT git add (nothing staged) ##########
$ git commit -m "Try commit without add"

	modified:   notes.txt

  (use "git add <file>..." to include in what will be committed)
	extra.txt

no changes added to commit (use "git add" and/or "git commit -a")

########## 1b. git commit -a -m (stages tracked notes.txt only) ##########
$ git commit -a -m "Update notes.txt using -a"
[main 4be685c] Update notes.txt using -a
 1 file changed, 1 insertion(+)

$ git status --short
?? extra.txt

$ git show --stat --oneline HEAD

########## 1c. untracked extra.txt still needs git add ##########
$ git add extra.txt
