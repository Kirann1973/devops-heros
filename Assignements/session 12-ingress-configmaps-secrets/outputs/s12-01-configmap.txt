$ (change LOG_LEVEL to DEBUG: the mounted file updates by itself, the env var keeps the old value)
$ kubectl patch configmap yatri-app-config -p '{"data":{"LOG_LEVEL":"DEBUG"}}'
configmap/yatri-app-config patched

$ (file changed after about 75 seconds)
$ kubectl exec configmap-demo -- cat /etc/config/LOG_LEVEL; echo

$ kubectl exec configmap-demo -- printenv LOG_LEVEL
INFO

$ kubectl delete pod configmap-demo --grace-period=1
